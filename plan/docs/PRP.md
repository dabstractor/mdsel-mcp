```typescript
import { McpServer } from '@modelcontextprotocol/server';

// CRITICAL: Use console.error for debug messages, NOT console.log
// In stdio mode, console.log corrupts the MCP protocol stream

// Minimal server that initializes but doesn't register tools yet
// Tools will be registered in P2.M2 (Implement Tool Handlers)

    name: 'mdsel-mcp',
    version: '1.0.0'
  },
  {
    capabilities: {
      tools: {}  // Enable tools capability
    }
  }
);

// For now, just prevent immediate exit
// In P2 we'll connect to stdio transport and handle requests

// Export for potential testing

```

```typescript
// File: tsup.config.ts - Build Configuration
import { defineConfig } from 'tsup';

export default defineConfig({
  // Single entry point for simple MCP server
  entry: ['src/index.ts'],

  // Output directory
  outDir: 'dist',

  // ES module format for modern Node.js
  format: ['esm'],

  // Target Node.js 18+ (MCP SDK requirement)
  target: 'node18',

  // Node.js platform (not browser)
  platform: 'node',

  // Generate TypeScript declaration files
  dts: true,

  // Clean dist directory before build
  clean: true,

  // CRITICAL: Add shebang for npx execution
  // This makes the built file executable directly
  banner: {
    js: '#!/usr/bin/env node',
  },

  // Don't split chunks - single file for simple CLI
  splitting: false,

  // Don't minify - easier debugging
  minify: false,

  // No sourcemaps in distribution
  sourcemap: false,
});
```

## Anti-Patterns to Avoid

- **Do NOT use console.log()** in src/index.ts - In stdio mode, this corrupts the MCP protocol. Use console.error() for debug output only.

- **Do NOT create complex directory structures** - The architecture specifies a single-file design (src/index.ts only). No src/handlers, src/utils, etc.

- **Do NOT add mdsel as a dependency** - It's a peer dependency. The MCP server assumes mdsel is installed globally or available in PATH.

- **Do NOT implement tool handlers yet** - This task is only package structure. Tool implementation happens in P2.M2.

- **Do NOT use CommonJS syntax** - No require(), module.exports. Use import/export exclusively.

- **Do NOT forget .js extensions** - In ESM, import paths must use .js even for .ts source files.

- **Do NOT minify in production** - Set minify: false for easier debugging.

- **Do NOT skip the shebang** - Without banner.js in tsup.config.ts, npx execution will fail.

- **Do NOT use old MCP SDK patterns** - The research documents v2 patterns using McpServer class, not the v1 Server class.

- **Do NOT commit dist/ or node_modules/** - These must be in .gitignore.
